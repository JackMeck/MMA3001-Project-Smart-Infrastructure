1"""Appendix study: does the sensor-to-room pairing matter? 

2 

3The sensor is assumed to belong to one room. This module repeats the analysis on 

4EVERY floorspace in the occupancy file, using the same chronological split and the 

5same (already chosen) final models with no re-tuning, to show how sensitive the 

6results are to that assumption. It does not prove which room the sensor is in. 

7""" 

8 

9from typing import Callable, Dict

11import pandas as pd 

12from scipy.stats import spearmanr 

13from sklearn.dummy import DummyRegressor 

14 

15from .config import Config 

16from .data import build_table, events_to_occupancy, read_events

20def occupancy_all_floorspaces(occ_csv: str, cfg: Config) -> Dict[str, pd.Series]: 

21 """Gridded occupancy of every floorspace in the study period. 

22 

23 Args: 

24 occ_csv: Path of the occupancy CSV. 

25 cfg: Configuration (``scan_start``, ``scan_end``). 

26 

27 Returns: 

28 Dict floorspace id -> occupancy series. Floorspaces with no valid events are 

29 skipped. 

30 """ 

31 ev = read_events(occ_csv, cfg, date_range=(cfg.scan_start, cfg.scan_end)) 

32 out = {} 

33 for fs, g in ev.groupby("floorspaceid"): 

34 try: 

35 out[fs] = events_to_occupancy(g, cfg) 

36 except ValueError: 

37 continue 

38 return out

41def floorspace_scan(env: pd.DataFrame, occ_all: Dict[str, pd.Series], cfg: Config, 

42 train_end) -> pd.DataFrame: 

43 """Rank floorspaces by CO2-occupancy association on the TRAINING period only. 

44 

45 Args: 

46 env: Gridded sensor frame. 

47 occ_all: Output of :func:`occupancy_all_floorspaces`. 

48 cfg: Configuration (``scan_min_train``). 

49 train_end: Last timestamp of the training period. 

50 

51 Returns: 

52 Frame indexed by the first 8 characters of the floorspace id with training 

53 rows, mean occupancy and the Spearman correlation between CO2 and 

54 occupancy, sorted strongest first. 

55 """ 

56 rows = [] 

57 for fs, occ in occ_all.items(): 

58 try: 

59 tr = build_table(env, occ).loc[:train_end] 

60 except ValueError: 

61 continue 

62 if len(tr) < cfg.scan_min_train: 

63 continue 

64 rows.append({"floorspace": fs[:8], "train rows": len(tr), 

65 "mean occupancy": tr["occupancy"].mean(), 

66 "Spearman(CO2, occupancy)": spearmanr(tr["co2"], tr["occupancy"])[0]}) 

67 cols = ["floorspace", "train rows", "mean occupancy", "Spearman(CO2, occupancy)"] 

68 scan = pd.DataFrame(rows, columns=cols).set_index("floorspace") 

69 return scan.sort_values("Spearman(CO2, occupancy)", ascending=False)

72def floorspace_sensitivity(env: pd.DataFrame, occ_all: Dict[str, pd.Series], 

73 models: Dict[str, Callable], features, cfg: Config, 

74 train_end, test_start) -> pd.DataFrame: 

75 """Fit and test the final models on every floorspace (same split, no re-tuning). 

76 

77 Args: 

78 env: Gridded sensor frame. 

79 occ_all: Output of :func:`occupancy_all_floorspaces`. 

80 models: Final model factories. 

81 features: Input columns. 

82 cfg: Configuration (``scan_min_train``, ``scan_min_test``). 

83 train_end: Last timestamp of the training period. 

84 test_start: First timestamp of the test period. 

85 

86 Returns: 

87 Long frame with ``floorspace``, ``model``, ``test mean occ`` and metrics. 

88 """ 

89 rows = [] 

90 models = dict(models) 

91 models["Constant (median)"] = lambda: DummyRegressor(strategy="median") 

92 for fs, occ in occ_all.items(): 

93 try: 

94 d = build_table(env, occ) 

95 except ValueError: 

96 continue 

97 tr, te = d.loc[:train_end], d.loc[test_start:] 

98 if len(tr) < cfg.scan_min_train or len(te) < cfg.scan_min_test: 

99 continue 

100 X_a, y_a, X_b, y_b = tr[features], tr["occupancy"], te[features], te["occupancy"] 

101 for name, make in models.items(): 

102 rows.append({"floorspace": fs[:8], "model": name, "test mean occ": y_b.mean(), 

103 **metrics(y_b, make().fit(X_a, y_a).predict(X_b))}) 

104 return pd.DataFrame(rows)
