73. sensitivity studies (feature ablation, SVR settings, training-data requirement); 

84. final models, cross-validation, computational cost, ONE final test, plots; 

95. appendix: repeat on every floorspace.

54def run_pipeline(cfg: Config, env_csv: str, occ_csv: str, out_dir: Optional[str] = None, 

55 appendix: bool = True, timing_repeats: int = 5, 

56 verbose: bool = False) -> Dict: 

57 """Run the whole analysis and return every result table. 

58 

59 Args: 

60 cfg: Configuration. 

61 env_csv: Path of the environmental CSV. 

62 occ_csv: Path of the occupancy CSV. 

63 out_dir: If given, tables are saved as CSV, the test plot as PNG and the 

64 settings/audit as JSON in this folder (created if needed). 

65 appendix: Also run the all-floorspace appendix study (slow: reads the whole 

66 occupancy file again). 

67 timing_repeats: Repeats for the timing study (best is reported). 

68 verbose: Print each table as it is produced. 

69 

70 Returns: 

71 Dict with keys ``audit``, ``table``, ``split``, ``cv_candidates``, 

72 ``ablation``, ``leave_one_out``, ``svr_grid``, ``svr_best``, ``curve_linear``, 

73 ``curve_final``, ``cv_final``, ``cost``, ``test``, ``preds``, ``improvement``, 

74 ``naive`` and, when ``appendix`` is True, ``scan`` and 

75 ``floorspace_sensitivity``. 

76 

77 Raises: 

78 ValueError: If the data are unusable (no CO2, no occupancy, no overlap, or too 

79 few rows to split). 

80 """ 

81 # ---- 1. data ------------------------------------------------------------------ 

82 env, audit = load_environment(env_csv, cfg) 

83 occ, n_events = load_occupancy(occ_csv, cfg) 

84 audit["occupancy_events"] = n_events 

85 audit["occupancy_rows"] = int(occ.notna().sum()) 

86 df = build_table(env, occ) 

87 sensor_cols = sensor_columns(env) 

88 features = sensor_cols + ["hour", "weekend"] 

89 X, y = df[features], df["occupancy"] 

90 audit["table_shape"] = list(df.shape) 

91 _say(verbose, "DATA AUDIT", json.dumps(audit, indent=2, default=str)) 

92 _say(verbose, "MODELLING TABLE SUMMARY", df.describe()) 

93 

94 # ---- 2. split + first comparison ------------------------------------------------ 

95 X_tr, y_tr, X_te, y_te = chronological_split(X, y, cfg) 

96 split = {"train": [str(X_tr.index.min()), str(X_tr.index.max())], 

97 "test": [str(X_te.index.min()), str(X_te.index.max())]} 

98 cv_candidates = cross_validate_models(candidate_models(sensor_cols), X_tr, y_tr, cfg) 

99 _say(verbose, "CROSS-VALIDATION (training period, mean of folds)", cv_candidates) 

100 

101 # ---- 3. sensitivity / optimisation (training period only) ------------------------- 

102 ablation = feature_ablation(X_tr, y_tr, X_te, y_te, features, sensor_cols, cfg) 

103 loo = leave_one_out(X_tr, y_tr, features, cfg) 

104 grid, best = svr_grid(X_tr, y_tr, cfg) 

105 curve_linear = training_size_curve(candidate_models(sensor_cols)["Linear Regression"], 

106 X_tr, y_tr, cfg) 

107 models = final_models(cfg) 

108 curve_final = data_requirement( 

109 {k: v for k, v in models.items() if k != "Baseline (schedule)"}, X_tr, y_tr, cfg) 

110 for title, obj in [("FEATURE ABLATION (linear model)", ablation), 

111 ("LEAVE-ONE-OUT (full linear model, CV)", loo), 

112 ("SVR (lean inputs): CV MAE, C (rows) x epsilon (columns)", grid), 

113 ("LINEAR MODEL: validation error vs training data", curve_linear), 

114 ("DATA REQUIREMENT: FINAL MODELS", curve_final)]: 

115 _say(verbose, title, obj) 

116 

117 # ---- 4. final evaluation ---------------------------------------------------------- 

118 cv_final = cross_validate_models(models, X_tr, y_tr, cfg) 

119 cost = computational_cost(models, X_tr, y_tr, X_te, timing_repeats) 

120 test, preds = final_test(models, X_tr, y_tr, X_te, y_te) 

121 improvement = mae_improvement(test) 

122 naive = naive_benchmarks(X_tr, y_tr, X_te, y_te) 

123 for title, obj in [("CROSS-VALIDATION OF FINAL MODELS", cv_final), 

124 ("COMPUTATIONAL COST", cost), ("FINAL TEST", test), 

125 ("MAE IMPROVEMENT OVER BASELINE", improvement), 

126 ("NAIVE CONSTANT BENCHMARKS (test set)", naive)]: 

127 _say(verbose, title, obj) 

128 

129 result = {"audit": audit, "table": df, "split": split, "cv_candidates": cv_candidates, 

130 "ablation": ablation, "leave_one_out": loo, "svr_grid": grid, 

131 "svr_best": best, "curve_linear": curve_linear, "curve_final": curve_final, 

132 "cv_final": cv_final, "cost": cost, "test": test, "preds": preds, 

133 "improvement": improvement, "naive": naive} 

134 

135 # ---- 5. appendix ------------------------------------------------------------------ 

136 if appendix: 

137 occ_all = occupancy_all_floorspaces(occ_csv, cfg) 

138 train_end, test_start = X_tr.index.max(), X_te.index.min() # identical split 

139 result["scan"] = floorspace_scan(env, occ_all, cfg, train_end) 

140 result["floorspace_sensitivity"] = floorspace_sensitivity( 

141 env, occ_all, models, features, cfg, train_end, test_start) 

142 _say(verbose, "FLOORSPACE SCAN (training period only)", result["scan"]) 

143 

144 if out_dir: 

145 save_results(result, cfg, out_dir, y_te) 

146 return result

149def save_results(result: Dict, cfg: Config, out_dir: str, y_te: pd.Series) -> None: 

150 """Write result tables (CSV), the test plot (PNG) and settings/audit (JSON). 

151 

152 Args: 

153 result: Output of :func:`run_pipeline`. 

154 cfg: Configuration used for the run. 

155 out_dir: Output folder (created if needed). 

156 y_te: Observed test headcount (for the plot). 

157 """ 

158 from dataclasses import asdict 

159 

160 os.makedirs(out_dir, exist_ok=True) 

161 for name, obj in result.items(): 

162 if isinstance(obj, (pd.DataFrame, pd.Series)) and name != "table": 

163 obj.to_csv(os.path.join(out_dir, f"{name}.csv")) 

164 with open(os.path.join(out_dir, "run_info.json"), "w") as fh: 

165 json.dump({"config": asdict(cfg), "audit": result["audit"], 

166 "split": result["split"], "svr_best": list(result["svr_best"])}, 

167 fh, indent=2, default=str) 

168 plot_test_results(result["test"], result["preds"], y_te, 

169 path=os.path.join(out_dir, "test_results.png"))

172def main(argv=None) -> None: 

173 """Command-line entry point (``python -m smart_infrastructure.pipeline``).""" 

174 p = argparse.ArgumentParser(description="Room-occupancy nowcasting workflow.") 

175 p.add_argument("--data-dir", required=True, 

176 help="Folder holding the two CSV files (or the .zip archives).") 

177 p.add_argument("--zip-dir", help="Folder with *Sensor*.zip files to extract first.") 

178 p.add_argument("--out", default="results", help="Output folder.") 

179 p.add_argument("--no-appendix", action="store_true", help="Skip the all-floorspace study.") 

180 a = p.parse_args(argv) 

181 cfg = Config() 

182 if a.zip_dir: 

183 unzip_data(a.zip_dir, a.data_dir) 

184 run_pipeline(cfg, os.path.join(a.data_dir, cfg.env_csv_name), 

185 os.path.join(a.data_dir, cfg.occ_csv_name), out_dir=a.out, 

186 appendix=not a.no_appendix, verbose=True)

189if __name__ == "__main__": 

190 main()
