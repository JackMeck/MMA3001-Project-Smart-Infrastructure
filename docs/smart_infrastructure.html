smart_infrastructure

Smart Infrastructure: nowcasting room occupancy from environmental sensor data.

The package estimates the number of people in a room (a nowcast of current occupancy) from environmental sensor readings (CO2, temperature, humidity, VOC, PM2.5) collected by the Monash Smart Infrastructure sensors.

Modules:

config: ~smart_infrastructure.config.Config dataclass holding every dataset-specific setting. data: parsing, cleaning, resampling and pairing of environmental and occupancy data. features: causal (trailing-window) feature construction. models: baselines and candidate scikit-learn regression models. evaluation: splits, cross-validation, metrics, sensitivity studies and profiling. pipeline: end-to-end workflow (~smart_infrastructure.pipeline.run_monash()).

Example:
>>> from smart_infrastructure import monash_config
>>> from smart_infrastructure.pipeline import run_monash
>>> cfg = monash_config(env_sensor_id="6012002000869",
...                     floorspace_id="941f0352-bb69-47d0-a9c5-ac29eb4f7319")
>>> result = run_monash(cfg, out_dir="results/run1")  # doctest: +SKIP

 38def build_features(df: pd.DataFrame, cfg: Config) -> FeatureSet:
 39    """Build trailing-window (causal) features from a clean regular-grid frame.
 40
 41    Feature groups:
 42        * ``raw``: the current reading of each environmental sensor.
 43        * ``rolling``: trailing mean and standard deviation over each window.
 44        * ``trend``: change over each window, ``x(t) - x(t - w)`` (e.g. CO2 rise rate).
 45        * ``time``: sine/cosine time-of-day and weekend flag (optional).
 46
 47    Args:
 48        df: Output of :func:`occupancy.io.clean`.
 49        cfg: Workflow configuration.
 50
 51    Returns:
 52        A :class:`FeatureSet`.
 53
 54    Raises:
 55        ValueError: If an excluded column is also listed as an input, or if a
 56            window is shorter than one grid step.
 57    """
 58    leak = set(cfg.exclude_cols) & set(cfg.env_cols)
 59    if leak:
 60        raise ValueError(f"Occupancy-derived columns used as inputs (leakage): {sorted(leak)}")
 61    step = cfg.step_minutes
 62    steps = []
 63    for w in cfg.windows_min:
 64        n = int(round(w / step))
 65        if n < 1:
 66            raise ValueError(f"Window {w} min is shorter than the {step} min grid step.")
 67        steps.append((w, n))
 68
 69    d = df.sort_values(["room", "timestamp"]).reset_index(drop=True)
 70    feats: Dict[str, pd.Series] = {}
 71    groups: Dict[str, List[str]] = {"raw": [], "rolling": [], "trend": [], "time": []}
 72    g = d.groupby("room", sort=False)
 73    for c in cfg.env_cols:
 74        feats[c] = d[c]
 75        groups["raw"].append(c)
 76        for w, n in steps:
 77            roll = g[c].transform(lambda s, n=n: s.rolling(n, min_periods=1).mean())
 78            diff = d[c] - g[c].shift(n)
 79            items = [(f"{c}_mean{w}m", roll, "rolling"), (f"{c}_chg{w}m", diff, "trend")]
 80            if n >= 2:  # a standard deviation needs at least two samples in the window
 81                std = g[c].transform(lambda s, n=n: s.rolling(n, min_periods=2).std())
 82                items.append((f"{c}_std{w}m", std, "rolling"))
 83            for name, ser, grp in items:
 84                feats[name] = ser
 85                groups[grp].append(name)
 86
 87    if cfg.use_time_features:
 88        hour = d["timestamp"].dt.hour + d["timestamp"].dt.minute / 60.0
 89        feats["hour_sin"] = np.sin(2 * np.pi * hour / 24.0)
 90        feats["hour_cos"] = np.cos(2 * np.pi * hour / 24.0)
 91        feats["is_weekend"] = (d["timestamp"].dt.dayofweek >= 5).astype(float)
 92        groups["time"] = ["hour_sin", "hour_cos", "is_weekend"]
 93
 94    unknown = set(cfg.feature_groups) - {"raw", "rolling", "trend", "time"}
 95    if unknown:
 96        raise ValueError(f"Unknown feature groups: {sorted(unknown)}")
 97    drop = [c for g, cols in groups.items() if g not in cfg.feature_groups for c in cols]
 98    groups = {g: cols for g, cols in groups.items() if g in cfg.feature_groups}
 99    if not groups:
100        raise ValueError("No feature groups selected.")
101    X = pd.DataFrame(feats).drop(columns=drop)
102    meta = d[["timestamp", "room"]].copy()
103    y = d[cfg.target_col].astype(float)
104    groups = {k: v for k, v in groups.items() if v}
105    return FeatureSet(X=X, y=y, meta=meta, groups=groups)

 65        cfg_seed: Random seed for stochastic models.
 66        baseline_col: Single raw sensor column for the simple physical baseline.
 67        include_schedule: Add the time-of-week-only ``baseline_schedule`` (needs the
 68            time features).
 69
 70    Returns:
 71        Ordered dict name -> unfitted Pipeline. Names starting with ``baseline_``
 72        are reference models, not selection candidates.
 73    """
 74    models = {
 75        "baseline_mean": Pipeline(
 76            _prep(False) + [("model", DummyRegressor(strategy="mean"))]
 77        ),
 78        "baseline_co2_linear": Pipeline(
 79            [("select", ColumnTransformer([("c", "passthrough", [baseline_col])])),
 80             *_prep(False), ("model", LinearRegression())]
 81        ),
 82        "linear": Pipeline(_prep(True) + [("model", Ridge(alpha=1.0))]),
 83        "tree": Pipeline(
 84            _prep(False)
 85            + [("model", DecisionTreeRegressor(max_depth=8, min_samples_leaf=20,
 86                                               random_state=cfg_seed))]
 87        ),
 88        "svr": Pipeline(_prep(True) + [("model", SVR(kernel="rbf", C=10.0, epsilon=0.5))]),
 89        "nn": Pipeline(
 90            _prep(True)
 91            + [("model", MLPRegressor(hidden_layer_sizes=(32, 16), early_stopping=True,
 92                                      max_iter=800, random_state=cfg_seed))]
 93        ),
 94    }
 95    if include_schedule:
 96        sched = Pipeline([
 97            ("select", ColumnTransformer([("t", "passthrough", TIME_COLS)])),
 98            ("model", ScheduleBaseline()),
 99        ])
100        out = {"baseline_mean": models.pop("baseline_mean"), "baseline_schedule": sched}
101        out.update(models)
102        return out
103    return models

def run_from_clean( cfg: Config, out_dir: str, clean_df: pandas.DataFrame, report: Dict, clean_builder=None, quick: bool = False) -> Dict: 
 39def run_from_clean(cfg: Config, out_dir: str, clean_df: pd.DataFrame, report: Dict,
 40                   clean_builder=None, quick: bool = False) -> Dict:
 41    """Features -> model selection -> final test -> studies, from a clean grid frame.
 42
 43    The held-out test period is scored exactly once, after model selection.
 44
 45    Args:
 46        cfg: Workflow configuration.
 47        out_dir: Output directory (created if needed).
 48        clean_df: Regular-grid frame ``timestamp, room, <target>, <env cols>``.
 49        report: Audit dictionary written to ``audit.json``.
 50        clean_builder: Optional ``cfg -> clean frame`` used by the resolution study.
 51        quick: Skip the slow resolution sweep and performance repeats.
 52
 53    Returns:
 54        Summary dictionary (selected model, final test metrics, paths).
 55    """
 56    os.makedirs(os.path.join(out_dir, "figures"), exist_ok=True)
 57    cfg.to_json(os.path.join(out_dir, "config_used.json"))
 58    with open(os.path.join(out_dir, "audit.json"), "w") as fh:
 59        json.dump(report, fh, indent=2, default=str)
 60
 61    # 1. features + splits
 62    fs = build_features(clean_df, cfg)
 63    tr, te = chronological_split(fs.meta, cfg.test_fraction, cfg.cv_gap_min)
 64    Xtr = fs.X.iloc[tr].reset_index(drop=True)
 65    ytr = fs.y.iloc[tr].reset_index(drop=True)
 66    folds = time_block_folds(fs.meta["timestamp"].iloc[tr], cfg.cv_splits, cfg.cv_gap_min)
 67
 68    # 3. model selection on training folds only
 69    models = make_models(cfg.seed, baseline_col=cfg.env_cols[0],
 70                         include_schedule="time" in fs.groups)
 71    if cfg.env_cols[0] not in fs.X.columns:       # CO2-only baseline needs the raw group
 72        models.pop("baseline_co2_linear")
 73    grids = param_grids()
 74    scoring = "neg_root_mean_squared_error" if cfg.selection_metric == "rmse" \
 75        else "neg_mean_absolute_error"
 76    tuned = {}
 77    for name, est in models.items():
 78        if name in grids:
 79            gs = GridSearchCV(est, grids[name], cv=folds, scoring=scoring, n_jobs=-1)
 80            gs.fit(Xtr, ytr)
 81            tuned[name] = clone(est).set_params(**gs.best_params_)
 82        else:
 83            tuned[name] = est
 84    cv = cross_validate(tuned, Xtr, ytr, folds, cfg)
 85    cv.to_csv(os.path.join(out_dir, "cv_folds.csv"), index=False)
 86    summary = cv.groupby("model").agg(["mean", "std"]).drop(columns="fold", level=0)
 87    summary.columns = ["_".join(c) for c in summary.columns]
 88    summary.to_csv(os.path.join(out_dir, "cv_summary.csv"))
 89    cands = summary.loc[[m for m in summary.index if not m.startswith("baseline_")]]
 90    best = cands[f"{cfg.selection_metric}_mean"].idxmin()
 91
 92    # 4. final test: fit on all training data, score ONCE
 93    final = clone(tuned[best]).fit(Xtr, ytr)
 94    Xte, yte = fs.X.iloc[te], fs.y.iloc[te]
 95    rows = []
 96    for name in tuned:
 97        m = final if name == best else clone(tuned[name]).fit(Xtr, ytr)
 98        rows.append({"model": name, "selected": name == best,
 99                     **regression_metrics(yte, m.predict(Xte), cfg)})
100    pd.DataFrame(rows).to_csv(os.path.join(out_dir, "final_test_metrics.csv"), index=False)
101    pred = final.predict(Xte)
102    pd.DataFrame({"timestamp": fs.meta["timestamp"].iloc[te].to_numpy(),
103                  "room": fs.meta["room"].iloc[te].to_numpy(),
104                  "observed": yte.to_numpy(),
105                  "predicted": np.clip(pred, 0, None)}).to_csv(
106        os.path.join(out_dir, "test_predictions.csv"), index=False)
107    for k, t in residual_tables(fs, te, pred, cfg).items():
108        t.to_csv(os.path.join(out_dir, f"residuals_{k}.csv"), index=False)
109    plot_diagnostics(yte, pred, fs.meta["timestamp"].iloc[te],
110                     os.path.join(out_dir, "figures", "final_test_diagnostics.png"),
111                     title=f"Final test - {best}")
112
113    # 5. grouped generalisation (only meaningful with >= 2 rooms)
114    if fs.meta["room"].nunique() >= 2:
115        g_rows = []
116        for tr_i, va_i in leave_one_room_out(fs.meta):
117            m = clone(tuned[best]).fit(fs.X.iloc[tr_i], fs.y.iloc[tr_i])
118            g_rows.append({"held_out_room": fs.meta["room"].iloc[va_i[0]],
119                           **regression_metrics(fs.y.iloc[va_i], m.predict(fs.X.iloc[va_i]), cfg)})
120        pd.DataFrame(g_rows).to_csv(os.path.join(out_dir, "leave_one_room_out.csv"), index=False)
121
122    # 6. sensitivity / optimisation (training folds; noise study post-selection)
123    feature_group_ablation(tuned[best], fs, tr, folds, cfg).to_csv(
124        os.path.join(out_dir, "sens_feature_groups.csv"), index=False)
125    training_size_curve(tuned[best], Xtr, ytr, folds, cfg).to_csv(
126        os.path.join(out_dir, "sens_training_size.csv"), index=False)
127    if "tree" in tuned:
128        hyperparameter_sweep(tuned["tree"], "model__max_depth", [2, 3, 5, 8, 12, 16],
129                             Xtr, ytr, folds, cfg).to_csv(
130            os.path.join(out_dir, "sens_tree_depth.csv"), index=False)
131    test_start = fs.meta["timestamp"].iloc[te].min()
132    noise_robustness(final, clean_df, cfg, test_start).to_csv(
133        os.path.join(out_dir, "sens_noise.csv"), index=False)
134    if not quick and clean_builder is not None:
135        resolution_sweep(clean_builder, tuned[best], cfg,
136                         rules=tuple(cfg.resolution_rules)).to_csv(
137            os.path.join(out_dir, "sens_resolution.csv"), index=False)
138
139    # 7. performance
140    performance_table(tuned, Xtr, ytr, Xte, repeats=1 if quick else 3).to_csv(
141        os.path.join(out_dir, "performance.csv"), index=False)
142
143    sel = next(r for r in rows if r["selected"])
144    result = {"selected_model": best, "n_train": int(len(tr)), "n_test": int(len(te)),
145              "final_test": sel, "out_dir": out_dir,
146              "train_period": [str(fs.meta["timestamp"].iloc[tr].min()),
147                               str(fs.meta["timestamp"].iloc[tr].max())],
148              "test_period": [str(fs.meta["timestamp"].iloc[te].min()),
149                              str(fs.meta["timestamp"].iloc[te].max())]}
150    for ref in ("baseline_mean", "baseline_schedule"):
151        r0 = next((r for r in rows if r["model"] == ref), None)
152        if r0 is not None and r0["mae"] > 0:
153            result[f"mae_improvement_vs_{ref}"] = 1.0 - sel["mae"] / r0["mae"]
154    with open(os.path.join(out_dir, "run_summary.json"), "w") as fh:
155        json.dump(result, fh, indent=2, default=float)
156    return result

def pairing_scan( cfg: Config, df_env: Optional[pandas.DataFrame] = None, occ: Optional[pandas.DataFrame] = None, floorspace_ids: Optional[List[str]] = None) -> pandas.DataFrame: 
159def pairing_scan(cfg: Config, df_env: Optional[pd.DataFrame] = None,
160                 occ: Optional[pd.DataFrame] = None,
161                 floorspace_ids: Optional[List[str]] = None) -> pd.DataFrame:
162    """Compare one sensor against several candidate floorspaces (training data only).
163
164    For each floorspace an environment-only decision tree (no time-of-day features)
165    is cross-validated on the training period. The final test period is never used.
166    A floorspace that really shares the sensor's room should be predicted much better
167    than the others. Every floorspace follows a similar weekday timetable, so raw
168    correlations cannot separate them; this scan can, but it is evidence, not proof.
169    Confirm with the official sensor-to-room mapping.
170
171    Args:
172        cfg: Configuration (``env_sensor_id`` and the candidate ``floorspace_ids``).
173        df_env: Raw environmental frame (read from ``cfg.env_path`` if None).
174        occ: Raw occupancy frame (read from ``cfg.occupancy_path`` if None).
175        floorspace_ids: Candidates (defaults to ``cfg.floorspace_ids``).
176
177    Returns:
178        One row per floorspace, best first: rows, Spearman(co2, occupancy), CV R2 and
179        MAE of the environment-only tree, and CV MAE of the schedule-only baseline.
180    """
181    ids = [str(f) for f in (floorspace_ids or cfg.floorspace_ids)]
182    if not ids:
183        raise ValueError("Give floorspace_ids to scan.")
184    if df_env is None:
185        df_env = pd.read_csv(cfg.env_path, usecols=["sensorid", "jsondata"])
186    ev_all = (to_events(occ, cfg, ids) if occ is not None
187              else read_occupancy(cfg.occupancy_path, cfg, ids))
188    env_grid, _, _ = prepare_env(df_env, cfg)
189    models = make_models(cfg.seed, cfg.env_cols[0], include_schedule=cfg.use_time_features)
190    rows = []
191    for fid in ids:
192        ev = ev_all[ev_all["floorspaceid"] == fid]
193        row = {"floorspace": fid[:8], "n_rows": 0}
194        try:
195            d = build_dataset(env_grid, occupancy_to_grid(ev, cfg), cfg, fid[:8])
196        except ValueError:
197            rows.append(row)
198            continue
199        fs = build_features(d, cfg)
200        tr, _ = chronological_split(fs.meta, cfg.test_fraction, cfg.cv_gap_min)
201        folds = time_block_folds(fs.meta["timestamp"].iloc[tr], cfg.cv_splits, cfg.cv_gap_min)
202        y = fs.y.iloc[tr].reset_index(drop=True)
203        X_all = fs.X.iloc[tr].reset_index(drop=True)
204        env_only = [c for g, v in fs.groups.items() if g != "time" for c in v]
205        res = cross_validate({"tree": models["tree"]}, X_all[env_only], y, folds, cfg)
206        row.update({"n_rows": int(len(fs.y)),
207                    "spearman_co2": float(X_all[cfg.env_cols[0]].corr(y, method="spearman")),
208                    "cv_r2_env_only": float(res["r2"].mean()),
209                    "cv_mae_env_only": float(res["mae"].mean())})
210        if "baseline_schedule" in models:
211            b = cross_validate({"s": models["baseline_schedule"]}, X_all, y, folds, cfg)
212            row["cv_mae_schedule_only"] = float(b["mae"].mean())
213        rows.append(row)
214    out = pd.DataFrame(rows)
215    return out.sort_values("cv_r2_env_only", ascending=False, na_position="last"
216                           ).reset_index(drop=True)
